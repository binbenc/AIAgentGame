import { expect, test } from 'testkit'
import { compareDates, daysInMonth, isLeapYear, parseDate } from '../src/date'

test('普通闰年', () => {
  expect(isLeapYear(2024)).toBe(true)
  expect(isLeapYear(2023)).toBe(false)
})

test('整百年不是闰年', () => {
  expect(isLeapYear(1900)).toBe(false)
  expect(isLeapYear(2100)).toBe(false)
})

test('每月天数', () => {
  expect(daysInMonth(2023, 2)).toBe(28)
  expect(daysInMonth(2024, 2)).toBe(29)
  expect(daysInMonth(2024, 4)).toBe(30)
  expect(daysInMonth(2024, 12)).toBe(31)
})

test('解析日期', () => {
  expect(parseDate('2024-03-15')).toEqual({ year: 2024, month: 3, day: 15 })
})

test('格式错误时报错', () => {
  expect(() => parseDate('2024/03/15')).toThrow('无效的日期格式')
})

test('月份越界时报错', () => {
  expect(() => parseDate('2024-13-01')).toThrow()
  expect(() => parseDate('2024-01-32')).toThrow()
})

test('比较日期', () => {
  expect(compareDates({ year: 2024, month: 1, day: 2 }, { year: 2024, month: 1, day: 1 }) > 0).toBe(true)
  expect(compareDates({ year: 2023, month: 12, day: 31 }, { year: 2024, month: 1, day: 1 }) < 0).toBe(true)
  expect(compareDates({ year: 2024, month: 5, day: 5 }, { year: 2024, month: 5, day: 5 })).toBe(0)
})
