/** A calendar date without a time zone. month starts at 1 (1 = January). */
export interface SimpleDate {
  year: number
  month: number
  day: number
}

/** Gregorian leap year: divisible by 4 but not by 100, or divisible by 400 */
export function isLeapYear(year: number): boolean {
  return (year % 4 === 0 && year % 100 !== 0) || year % 400 === 0
}

const DAYS = [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31]

/** Number of days in a given month */
export function daysInMonth(year: number, month: number): number {
  if (month === 2 && isLeapYear(year)) return 29
  return DAYS[month - 1]
}

/** Parse YYYY-MM-DD; throws on a bad format or a date that doesn't exist */
export function parseDate(s: string): SimpleDate {
  const m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(s)
  if (!m) throw new Error(`Invalid date format: ${s}`)
  const d = { year: Number(m[1]), month: Number(m[2]), day: Number(m[3]) }
  if (d.month < 1 || d.month > 12) throw new RangeError(`Invalid month: ${s}`)
  if (d.day < 1 || d.day > daysInMonth(d.year, d.month)) throw new RangeError(`Invalid date: ${s}`)
  return d
}

/** Negative if a is before b, 0 if equal, positive if a is after b */
export function compareDates(a: SimpleDate, b: SimpleDate): number {
  return a.year - b.year || a.month - b.month || a.day - b.day
}
