import { expect, test } from 'testkit'
import { checkout } from '../src/checkout'
import { applyCoupons } from '../src/coupons'

const F20 = { code: 'F20', kind: 'fixed' as const, value: 2000, minSpend: 10000 }

test('fixed coupon applies when the subtotal equals the minimum spend', () => {
  expect(applyCoupons(10000, [F20])).toEqual({ total: 8000, applied: ['F20'] })
})

test('checkout: exactly ¥100 gets ¥20 off', () => {
  const r = checkout([{ sku: 'BAG', name: 'Tote bag', unitPrice: 5000, quantity: 2 }], [F20])
  expect(r.total).toBe(8000)
  expect(r.coupons).toEqual(['F20'])
})
