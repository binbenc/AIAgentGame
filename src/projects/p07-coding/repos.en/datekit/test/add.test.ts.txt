import { expect, test } from 'testkit'
import { addDays, addMonths } from '../src/add'

test('add days', () => {
  expect(addDays({ year: 2024, month: 1, day: 30 }, 3)).toEqual({ year: 2024, month: 2, day: 2 })
})

test('add days across a year', () => {
  expect(addDays({ year: 2023, month: 12, day: 31 }, 1)).toEqual({ year: 2024, month: 1, day: 1 })
})

test('subtract days', () => {
  expect(addDays({ year: 2024, month: 3, day: 10 }, -5)).toEqual({ year: 2024, month: 3, day: 5 })
})

test('add months', () => {
  expect(addMonths({ year: 2024, month: 1, day: 15 }, 1)).toEqual({ year: 2024, month: 2, day: 15 })
})

test('clamp to end of month', () => {
  expect(addMonths({ year: 2023, month: 1, day: 31 }, 1)).toEqual({ year: 2023, month: 2, day: 28 })
  expect(addMonths({ year: 2024, month: 1, day: 31 }, 1)).toEqual({ year: 2024, month: 2, day: 29 })
})

test('add months across a year', () => {
  expect(addMonths({ year: 2024, month: 11, day: 30 }, 3)).toEqual({ year: 2025, month: 2, day: 28 })
})
