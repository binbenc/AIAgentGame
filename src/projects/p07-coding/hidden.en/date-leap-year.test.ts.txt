import { expect, test } from 'testkit'
import { addMonths } from '../src/add'
import { daysInMonth, isLeapYear } from '../src/date'

test('years divisible by 400 are leap years', () => {
  expect(isLeapYear(2000)).toBe(true)
  expect(isLeapYear(2400)).toBe(true)
})

test('February 2000 has 29 days', () => {
  expect(daysInMonth(2000, 2)).toBe(29)
})

test('2000-01-31 plus 1 month is 2000-02-29', () => {
  expect(addMonths({ year: 2000, month: 1, day: 31 }, 1)).toEqual({ year: 2000, month: 2, day: 29 })
})
