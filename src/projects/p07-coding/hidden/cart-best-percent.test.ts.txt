import { expect, test } from 'testkit'
import { applyCoupons } from '../src/coupons'

const P10 = { code: 'P10', kind: 'percent' as const, value: 10 }
const P20 = { code: 'P20', kind: 'percent' as const, value: 20 }

test('多张百分比券时选折扣最大的', () => {
  expect(applyCoupons(10000, [P10, P20])).toEqual({ total: 8000, applied: ['P20'] })
})

test('顺序无关', () => {
  expect(applyCoupons(10000, [P20, P10])).toEqual({ total: 8000, applied: ['P20'] })
})
