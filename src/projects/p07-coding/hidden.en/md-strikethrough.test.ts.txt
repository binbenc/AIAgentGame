import { expect, test } from 'testkit'
import { renderInline } from '../src/inline'
import { renderMarkdown } from '../src/block'

test('strikethrough', () => {
  expect(renderInline('~~old~~ new')).toBe('<del>old</del> new')
})

test('strikethrough in a paragraph', () => {
  expect(renderMarkdown('price: ~~100~~ 80')).toBe('<p>price: <del>100</del> 80</p>')
})
