import { expect, test } from 'testkit'
import { formatDate } from '../src/format'

test('pads Sep 9', () => {
  expect(formatDate({ year: 2024, month: 9, day: 9 })).toBe('2024-09-09')
})

test('pads MM/DD', () => {
  expect(formatDate({ year: 2024, month: 9, day: 30 }, 'MM/DD')).toBe('09/30')
  expect(formatDate({ year: 2024, month: 10, day: 9 }, 'MM/DD')).toBe('10/09')
})
