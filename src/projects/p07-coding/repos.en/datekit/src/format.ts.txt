import type { SimpleDate } from './date'

function pad2(n: number): string {
  return n < 10 ? `0${n}` : String(n)
}

/**
 * Format a date with a pattern. Supported tokens:
 * - YYYY four-digit year
 * - MM / DD two-digit month / day (zero-padded)
 * - M / D month / day without padding
 */
export function formatDate(d: SimpleDate, pattern = 'YYYY-MM-DD'): string {
  return pattern.replace(/YYYY|MM|DD|M|D/g, (token) => {
    switch (token) {
      case 'YYYY':
        return String(d.year).padStart(4, '0')
      case 'MM':
        return pad2(d.month)
      case 'DD':
        return pad2(d.day)
      case 'M':
        return String(d.month)
      default:
        return String(d.day)
    }
  })
}
